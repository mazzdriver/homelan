```javascript
// Мок для ssh2 (транспорт Admin)

                handler(JSON.stringify({
                  platform: 'linux',
                  distro: 'ubuntu',
                  version: '22.04',
                  init: 'systemd',
                  package_manager: 'apt',
                  capabilities: ['lock-screen', 'poweroff', 'autologin']
                }));
              } else if (cmd.includes('homelan-client plan')) {
                handler(JSON.stringify({ packages: ['light-locker', 'accountsservice'], reboot: false }));
              }
              handler(null); // EOF
            }
          }),
          end: jest.fn()
        };
        return cb(null, stream);
      }),
      end: jest.fn()
    }))
  };
});

// Мок для child_process (адаптеры Client)
const originalExecFile = jest.requireActual('child_process').execFile;
jest.mock('child_process', () => ({
  execFile: jest.fn((file, args, options, callback) => {
    // systemctl
    if (file === 'systemctl') {
      if (args.includes('is-active')) {
        return callback(null, { stdout: 'active\n' });
      }
      if (args.includes('start')) {
        return callback(null);
      }
    }
    // loginctl
    if (file === 'loginctl') {
      if (args.includes('lock-session')) {
        return callback(null);
      }
    }
    // uname
    if (file === 'uname') {
      if (args.includes('-s')) {
        return callback(null, { stdout: 'Linux\n' });
      }
    }
    // Win32 fallback
    return callback(new Error('ENOENT: no such file or directory'));
  })
}));
```

## Файл: `02-next/tests/protocol/lines.test.js`

```javascript
// 02-next/tests/protocol/lines.test.js
const { parseLines } = require('../../packages/homelan-protocol/src/lines.js');

describe('JSON-lines protocol resilience', () => {
  it('[P-01] should skip invalid JSON and continue stream', () => {
    const stream = [
      '{\"jsonrpc\":\"2.0\",\"method\":\"ping\"}\n',
      'not a json at all\n',
      '{\"jsonrpc\":\"2.0\",\"method\":\"pong\",\"id\":1}\n'
    ];
    
    const results = [];
    stream.forEach(line => {
      try {
        const msg = parseLines(line);
        if (msg) results.push(msg);
      } catch (e) {
        // parseLines может бросать, а может возвращать null.
        // Если бросает - это баг протокола, тест должен упасть.
      }
    });

    expect(results).toHaveLength(2);
    expect(results[0].method).toBe('ping');
    expect(results[1].method).toBe('pong');
  });

  it('[P-04] should handle empty lines and whitespace', () => {
    const inputs = [
      '\n',
      '   \n',
      '{\"jsonrpc\":\"2.0\",\"method\":\"test\"}\n',
      '\t\n'
    ];
    
    const results = inputs.map(line => {
      try {
        return parseLines(line);
      } catch {
        return null;
      }
    }).filter(r => r !== null);

    expect(results).toHaveLength(1);
    expect(results[0].method).toBe('test');
  });
});

```

## Файл: `02-next/tests/admin/lock.partial-fail.test.js`

```javascript
// 02-next/tests/admin/lock.partial-fail.test.js
const { execSync } = require('child_process');
const path = require('path');

// Перезатираем глобальный мок из setupTests.js для этого файла
// чтобы симулировать частичный отказ
jest.mock('ssh2', () => {
  return {
    Client: jest.fn(() => ({
      on: jest.fn(),
      connect: jest.fn((cfg, cb) => cb()),
      exec: jest.fn((cmd, opts, cb) => {
        const stream = {
          on: jest.fn((event, handler) => {
            if (event === 'data') {
              if (cmd.includes('newpass')) {
                // device-3 упадет на этапе newpass
                if (cmd.includes('device-3')) {
                  handler(JSON.stringify({ error: 'ERR_CHILD_PROCESS_FAIL' }));
                } else {
                  handler(JSON.stringify({ password: 'temp-123' }));
                }
                handler(null);
              }
            }
          }),
          end: jest.fn()
        };
        cb(null, stream);
      }),
      end: jest.fn()
    }))
  };
});

describe('homelan lock -- partial failure (A-05)', () => {
  it('should fail overall if one device fails to reset password', () => {
    const bin = path.resolve('../../packages/homelan-admin/bin/homelan.js');
    
    // Мокаем реестр устройств
    jest.mock('../../packages/homelan-admin/src/device-registry.js', () => ({
      loadRegistry: jest.fn(() => ({
        'device-1': { address: '10.0.0.1', user: 'p' },
        'device-2': { address: '10.0.0.2', user: 'p' },
        'device-3': { address: '10.0.0.3', user: 'p' }
      })),
      saveRegistry: jest.fn()
    }));
    
    try {
      execSync(`node "${bin}" lock device-1,device-2,device-3 --user child`, { encoding: 'utf-8' });
    } catch (err) {
      expect(err.status).toBe(1);
    }

    const registry = require('../../packages/homelan-admin/src/device-registry.js');
    // Операция должна быть атомарной: если один упал, пароли не сбрасываются нигде
    expect(registry.saveRegistry).not.toHaveBeenCalled();
  });
});
```
